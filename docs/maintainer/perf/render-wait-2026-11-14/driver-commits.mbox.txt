+owned.TEST_PACKAGE = TEST_PACKAGE = PACKAGE + '.test'

+CACHE_PATHS = tuple(f'/data/{storage}/0/{PACKAGE}/cache/com.android.{kind}.shaders_cache'
+                    for storage in ('user_de', 'user') for kind in ('skia', 'opengl'))

+def cache_script(clear=False):
+    paths = ' '.join(shlex.quote(path) for path in CACHE_PATHS)
+    # Refuse links/directories and never recursively remove a cache directory.
+    action = 'rm -- "$cache_file"' if clear else 'sha256sum "$cache_file"; wc -c < "$cache_file"'
+    return (f'for cache_file in {paths}; do '
+            'if [ -L "$cache_file" ]; then exit 72; fi; '
+            'if [ -e "$cache_file" ] && [ ! -f "$cache_file" ]; then exit 73; fi; '
+            f'if [ -f "$cache_file" ]; then {action}; fi; done')
+
+
+CACHE_COMMANDS = {clear: ('shell', 'run-as', PACKAGE, 'sh', '-c', shlex.quote(cache_script(clear)))
+                  for clear in (False, True)}

+def parse_cache(text):
+    lines = text.splitlines()
+    if len(lines) % 2:
+        raise ValueError('Incomplete shader cache inventory')
+    records = []
+    for index in range(0, len(lines), 2):
+        match = re.fullmatch(r'([a-f0-9]{64})\s+(.+)', lines[index])
+        if not match or match[2] not in CACHE_PATHS or not lines[index + 1].strip().isdigit():
+            raise ValueError('Unexpected shader cache inventory')
+        records.append(dict(path=match[2], sha256=match[1], bytes=int(lines[index + 1])))
+    if len({item['path'] for item in records}) != len(records):
+        raise ValueError('Duplicate shader cache entry')
+    return records
+

+class Device(owned.OwnedDevice):
+    driver_owned = False
+
+    def permit(self, parts):
+        p = tuple(map(str, parts))
+        if p in {('shell', 'test', '-e', DRIVER), ('shell', 'sha256sum', DRIVER)}:
+            return
+        if self.active and PACKAGE in self.owned:
+            if p in CACHE_COMMANDS.values():
+                return
+            if p == ('push', str(self.args.host.resolve()), DRIVER) and self.driver_owned:
+                return
+            if p in {('shell', 'chmod', '444', DRIVER), ('shell', 'rm', '--', DRIVER)} and self.driver_owned:
+                return
+        super().permit(parts)
+
+    def inventory(self):
+        return parse_cache(self.call(*CACHE_COMMANDS[False]).stdout)
+
+    def stop_product(self):

+            time.sleep(0.05)

+"""Validate the diagnostic's file/ownership boundary without contacting a device."""
+import importlib.util
+from pathlib import Path
+from types import SimpleNamespace
+import unittest
+
+spec = importlib.util.spec_from_file_location('render_cache_probe', Path(__file__).with_name('run-render-cache-probe.py'))

+class CacheProbeTest(unittest.TestCase):
+    def test_inventory_rejects_truncation_duplicates_and_unexpected_paths(self):
+        path = probe.CACHE_PATHS[0]
+        entry = 'a' * 64 + '  ' + path + '\n123\n'
+        self.assertEqual(probe.parse_cache(entry), [dict(path=path, sha256='a' * 64, bytes=123)])
+        for broken in (entry.splitlines()[0], entry + entry, entry.replace(path, '/data/other/file'),
+                       entry.replace('123', '-1'), entry.replace('a' * 64, 'x' * 64)):
+            with self.subTest(broken=broken), self.assertRaises(ValueError):
+                probe.parse_cache(broken)
+
+    def test_removal_requires_acquired_package_ownership(self):
+        device = probe.Device(SimpleNamespace(host=Path('/tmp/host.apk')), {})
+        for active, owned in ((False, set()), (True, set()), (True, {probe.owned.HOST})):
+            device.active = active
+            device.initial = {'wakefulness': 'Awake'}
+            device.owned = owned
+            with self.subTest(active=active, owned=owned), self.assertRaises(ValueError):
+                device.permit(probe.CACHE_COMMANDS[True])
+        device.owned = {probe.PACKAGE}
+        device.permit(probe.CACHE_COMMANDS[True])
+
+    def test_only_fixed_shader_files_and_owned_injector_are_allowed(self):
+        device = probe.Device(SimpleNamespace(host=Path('/tmp/host.apk')), {})
+        device.active = True
+        device.initial = {'wakefulness': 'Awake'}
