+package io.github.sandyyur.inputbench;
+
+import java.net.ServerSocket;
+import java.net.Socket;
+import java.net.InetAddress;
+import android.util.Log;
+import android.os.Process;
+import android.os.SystemClock;
+import android.view.InputDevice;
+import android.view.InputEvent;
+import android.view.MotionEvent;
+import java.io.BufferedReader;
+import java.io.InputStreamReader;
+import java.io.OutputStreamWriter;
+import java.io.PrintWriter;
+import java.lang.reflect.Method;
+import org.json.JSONArray;
+import org.json.JSONObject;
+
+/** Shell-only experiment driver: never loaded into the IME or shipped in its APK. */
+public final class BenchTouchDriver {
+    private static long now() { return SystemClock.elapsedRealtimeNanos(); }
+
+    private static void waitUntil(long time) {
+        for (;;) {
+            long remaining = time - now();
+            if (remaining <= 0) return;
+            SystemClock.sleep((remaining + 999_999L) / 1_000_000L);
+        }
+    }
+
+    public static void main(String[] args) throws Exception {
+        try { run(args); }
+        catch (Exception error) { Log.e("Pr32TouchDriver", "External input driver failed", error); throw error; }
+    }
+
+    private static void run(String[] args) throws Exception {
+        if (args.length != 1 || !args[0].matches("pr32-touch-[0-9]+"))
+            throw new IllegalArgumentException("Expected a dedicated experiment socket name");
+        Class<?> inputClass = Class.forName("android.hardware.input.InputManagerGlobal");
+        Object input = inputClass.getMethod("getInstance").invoke(null);
+        Method inject = inputClass.getMethod("injectInputEvent", InputEvent.class, int.class);
+        int port = 20000 + Integer.parseInt(args[0].substring("pr32-touch-".length())) % 20000;
+        ServerSocket server = new ServerSocket(port, 1, InetAddress.getByName("127.0.0.1"));
+        server.setSoTimeout(60_000);
+        Socket socket = null;
+        try {
+            socket = server.accept();
+            try (BufferedReader reader = new BufferedReader(new InputStreamReader(socket.getInputStream(), "UTF-8"));
+             PrintWriter writer = new PrintWriter(new OutputStreamWriter(socket.getOutputStream(), "UTF-8"), true)) {
+            writer.println(new JSONObject().put("pid", Process.myPid()).put("uid", Process.myUid())
+                .put("protocol", "external_input_manager_v1").put("token", args[0]).put("transport", "loopback_tcp")
+                .put("driver_version", 2).put("actual_hold_min_ns", 20_000_000L));
+            String line;
+            while ((line = reader.readLine()) != null) {
+                JSONObject request = new JSONObject(line);
+                JSONArray touches = request.getJSONArray("touches");
+                if (touches.length() < 1 || touches.length() > 2) throw new IllegalArgumentException("Invalid batch");
+                JSONArray results = new JSONArray();
+                for (int index = 0; index < touches.length(); index++) {
+                    JSONObject touch = touches.getJSONObject(index);
+                    long plannedDown = touch.getLong("planned_down_ns");
+                    long plannedUp = touch.getLong("planned_up_ns");
+                    if (plannedUp - plannedDown != 20_000_000L) throw new IllegalArgumentException("Invalid hold");
+                    long downTime = touch.getLong("down_time_ms");
+                    JSONObject result = new JSONObject().put("down_time_ms", downTime).put("key", touch.getString("key"));
+                    long actualDown = 0;
+                    for (int action : new int[]{MotionEvent.ACTION_DOWN, MotionEvent.ACTION_UP}) {
+                        // A late DOWN must not compress the intended physical hold to zero.
+                        waitUntil(action == MotionEvent.ACTION_DOWN ? plannedDown : Math.max(plannedUp, actualDown + 20_000_000L));
+                        MotionEvent event = MotionEvent.obtain(downTime, SystemClock.uptimeMillis(), action,
+                            (float)touch.getDouble("x"), (float)touch.getDouble("y"), 0);
+                        event.setSource(InputDevice.SOURCE_TOUCHSCREEN);
+                        boolean down = action == MotionEvent.ACTION_DOWN;
+                        long timestamp = now();
+                        if (down) actualDown = timestamp;
+                        result.put(down ? "inject_down_ns" : "inject_up_ns", timestamp);
+                        try {
+                            if (!Boolean.TRUE.equals(inject.invoke(input, event, 0)))
+                                throw new IllegalStateException("Input injection rejected");
+                        } finally { event.recycle(); }
+                        result.put(down ? "inject_down_return_ns" : "inject_up_return_ns", now());
+                    }
+                    results.put(result);
+                }
+                writer.println(new JSONObject().put("touches", results));
+                if (writer.checkError()) throw new IllegalStateException("Experiment client disconnected");
+            }
+            }
+        } finally {
+            if (socket != null) socket.close();
+            server.close();
+        }
+    }
+}

+def signer_sha256(report):
+    # New build-tools label scheme signers "V2 Signer:"; older versions use
+    # "Signer #1". One certificate may appear under multiple APK schemes.
+    certificates = {
+        match.lower() for match in re.findall(
+            r'^(?:Signer #\d+|V[1-4](?:\.\d+)? Signer:) certificate SHA-256 digest: ([a-fA-F0-9]{64})\s*$',
+            report, re.MULTILINE)
+    }
+    if len(certificates) != 1:
+        raise ValueError('Expected one distinct verified APK signing certificate')
+    return certificates.pop()

+def interval_evidence(start,end,utid,sched,states):
+    """A zero CPU result is meaningful only when the thread is observed throughout."""
+    clipped=[(max(start,ts),min(end,ts+dur),state)
+        for ts,dur,thread_id,state in states
+        if end>=start and thread_id==utid and ts<end and ts+dur>start]
+    state_ns={}
+    covered=0
+    cursor=start
+    for left,right,state in sorted(clipped):
+        state_ns[state]=state_ns.get(state,0)+right-left
+        covered+=max(0,right-max(cursor,left))
+        cursor=max(cursor,right)
+    cpu_ns=sum(max(0,min(end,ts+dur)-max(start,ts))
+        for ts,dur,thread_id in sched if thread_id==utid)
+    running_ns=state_ns.get('Running',0)
+    duration=end-start
+    # Union coverage detects gaps; the sum also rejects overlapping state records.
+    complete=(duration>=0 and covered==duration and sum(state_ns.values())==duration
+        and cpu_ns==running_ns)
+    return {'utid':utid,'state_covered_ns':covered,'interval_ns':duration,
+        'running_ns':running_ns,'sched_cpu_ns':cpu_ns,'complete':complete,
+        'cpu_ms':cpu_ns/1e6 if complete else None,
+        'state_ms':{state:ns/1e6 for state,ns in state_ns.items()}}

+"""Raw AndroidJUnitRunner status parser, shared by isolated quality checks."""
+def parse_status(text):
+    """Parse AndroidJUnitRunner raw status packets, including stack continuations."""
+    events, packet, last_key = [], {}, None
+    for line in text.splitlines():
+        if line.startswith('INSTRUMENTATION_STATUS: '):
+            field = line.removeprefix('INSTRUMENTATION_STATUS: ')
+            key, sep, value = field.partition('=')
+            if sep:
+                packet[key], last_key = value, key
+        elif line.startswith('INSTRUMENTATION_STATUS_CODE: '):
+            code = int(line.removeprefix('INSTRUMENTATION_STATUS_CODE: '))
+            if code != 1 and packet.get('test') and packet.get('class'):
+                events.append(dict(packet, status_code=code))
+            packet, last_key = {}, None
+        elif line.startswith('INSTRUMENTATION_'):
+            last_key = None
+        elif last_key:
+            packet[last_key] += '\n' + line
+    if 'INSTRUMENTATION_CODE: -1' not in text or 'INSTRUMENTATION_FAILED:' in text:
+        raise ValueError('The instrumentation process did not finish normally')
+    if not events:
+        raise ValueError('No executed test case results')
+    return events

+def digest(path):
+    result = hashlib.sha256()
+    with path.open('rb') as source:

+def apk_identity(apk, aapt, java, signer_jar):
+    tree = host_command((aapt, 'dump', 'xmltree', '--file', 'AndroidManifest.xml', apk))
+    package = re.findall(r'^\s*A: package="([^"]+)"', tree, re.MULTILINE)
+    # aapt2 may print the Android namespace URI instead of its short prefix.
+    namespace = r'(?:android:|http://schemas\.android\.com/apk/res/android:)'
+    def attributes(name):
+        prefix = r'A:\s+' + namespace + name
+        values = re.findall(prefix + r'(?:\([^)]*\))?="([^"]+)"', tree)
+        if len(values) != len(re.findall(prefix + r'(?=\(|=)', tree)):
+            raise ValueError(f'Cannot resolve {name} in {apk.name}')
+        return values
+    targets = attributes('targetPackage')
+    if (len(package) != 1 or len(targets) > 1 or
+            re.search(r'A:\s+' + namespace + r'sharedUserId(?=\(|=)', tree)):
+        raise ValueError(f'Unexpected package/UID manifest in {apk.name}')
+    authorities = attributes('authorities')
+    if any(not authority.startswith(package[0] + '.') for authority in authorities):
+        raise ValueError(f'Provider authority is outside its diagnostic package: {apk.name}')
+    signature = host_command((java, '-jar', signer_jar, 'verify', '--print-certs', apk))
+    from apk_signers import signer_sha256
+    certificate = signer_sha256(signature)
+    return {'package': package[0], 'target': targets[0] if targets else None,
+            'signer_sha256': certificate, 'authorities': authorities}

+        if state['default_ime'] != 'null' and not re.fullmatch(r'[A-Za-z0-9_.]+/[A-Za-z0-9_.]+', state['default_ime']):
+            raise RuntimeError('Cannot safely restore the current default IME')
+        if state['default_ime'] != 'null' and state['default_ime'].split('/', 1)[0] not in installed:
+            raise RuntimeError('The original default IME package is not installed')

+            raise RuntimeError('Unexpected settings values; refuse to change them')

+        self.args.output.mkdir(parents=True, exist_ok=True)

+    def power_state(self):
+        power_dump = self.call('shell', 'dumpsys', 'power').stdout
+        return {'wakefulness': parse_wakefulness(power_dump), 'power_dump': power_dump}

+    def wait_wakefulness(self, expected):
+        deadline = time.monotonic() + 10
+        while True:
+            state = self.power_state()
+            if state['wakefulness'] == expected:
+                return state
+            if time.monotonic() >= deadline:
+                raise RuntimeError(f'Device did not become {expected}; last state: {state["wakefulness"]}')

+    def wake_for_study(self):
+        if self.initial['wakefulness'] == 'Asleep':

+            self.call('shell', 'input', 'keyevent', 'KEYCODE_WAKEUP')
+            save(self.args.output / 'wake-up.json', self.wait_wakefulness('Awake'))

+    def restore_wakefulness(self, final):

+            key = 'KEYCODE_WAKEUP' if expected == 'Awake' else 'KEYCODE_SLEEP'
+            self.call('shell', 'input', 'keyevent', key)

+    def retain_failed_raw(self):
+        retained = []

+                entry.update(saved=False, error=repr(error))

+    def install(self, path):

+        if package in self.packages():
+            raise RuntimeError('Package appeared before fresh install; it is not owned: ' + package)

+                   'status': 'pending'}

+        # PackageManager has installed the APK but before adb reports success.
+        def record():

+                 'owned': sorted(self.owned), 'install_attempts': self.install_attempts})
+        record()

+            result = self.call('install', path, timeout=180)
+            if 'Success' not in result.stdout:
+                raise RuntimeError('Installation did not report success: ' + package)
+            self.owned.add(package)
+            attempt['status'] = 'confirmed'
+            record()
+        except BaseException as error:
+            if package not in self.owned:
+                attempt['status'] = 'unconfirmed'
+            attempt['error'] = repr(error)
+            try:
+                record()
+            except OSError as audit_error:
+                self.audit_errors.append(repr(audit_error))

+    def uninstall(self, package):
+        if package not in self.owned:
+            return

+        if 'Success' not in result.stdout or package in self.packages():
+            raise RuntimeError('Owned package removal failed: ' + package)

+    def clock(self):
+        value = self.call('shell', 'date', '+%s.%N').stdout.strip()
+        if not re.fullmatch(r'\d+\.\d+', value):
+            raise RuntimeError('Device clock cannot delimit this instrumentation run')
+        # logcat's epoch format may print only milliseconds. Keep the lower bound
+        # at that precision; the independently checked fresh PID rejects old runs.
+        return format(Decimal(value).quantize(Decimal('0.001'), rounding=ROUND_FLOOR), 'f')

+    def instrument(self, method, revision, prefix):

+                             'stdout': observed.stdout, 'stderr': observed.stderr})

+                raise RuntimeError('Previous diagnostic process did not stop')

+                 'org.fcitx.fcitx5.android.core.CandidateTapTest#' + method, '-e', 'expectedCommit', revision,
+                 TEST_PACKAGE + '/androidx.test.runner.AndroidJUnitRunner')

+        observed_pid = None

+            if method == 'replay':

+                started = False

+                    pid = self.call('shell', 'pidof', PACKAGE, check=False).stdout.strip()

+                        logs = self.call('logcat', '-d', '-v', 'epoch', '-T', checkpoint, '--pid=' + pid,
+                                         '-s', 'TestRunner:I', '*:S').stdout
+                        for line in logs.splitlines():
+                            match = re.match(r'^\s*(\d+\.\d+)\s+(\d+)\s+\d+\s+I\s+TestRunner\s*:\s*started: replay\(org\.fcitx\.fcitx5\.android\.core\.CandidateTapTest\)', line)
+                            if match and match[2] == pid and Decimal(match[1]) >= Decimal(checkpoint):
+                                started = True
+                                break
+                    if started:
+                        self.call('shell', 'ime', 'enable', COMPONENT)
+                        self.call('shell', 'ime', 'set', COMPONENT)
+                        if self.call('shell', 'settings', 'get', 'secure', 'default_input_method').stdout.strip() != COMPONENT:
+                            raise RuntimeError('The diagnostic IME was not selected')
+                        break
+                    time.sleep(0.1)
+                if not started:
+                    raise RuntimeError('No fresh replay start from the current diagnostic PID')
+            stdout, stderr = process.communicate(timeout=900)
+            prefix.with_suffix('.instrumentation.txt').write_text(stdout + stderr, encoding='utf-8')
+            if process.returncode != 0 or 'OK (1 test)' not in stdout:
+                raise RuntimeError('Instrumentation failed: ' + method)
+        except BaseException:
+            if process.poll() is None:
+                process.kill()
+            stdout, stderr = process.communicate(timeout=15)
+            if not prefix.with_suffix('.instrumentation.txt').exists():
+                prefix.with_suffix('.instrumentation.txt').write_text(stdout + stderr, encoding='utf-8')
+            raise
+        finally:
+            try:
+                pid = observed_pid or self.call('shell', 'pidof', PACKAGE, check=False).stdout.strip()

+                prefix.with_suffix('.logcat.txt').write_text(logs, encoding='utf-8')
+            except Exception as error:
+                prefix.with_suffix('.logcat-error.txt').write_text(repr(error), encoding='utf-8')

+    def cleanup(self):
+        self.restoring = True
+        errors = []
+        def attempt(name, action):
+            try:
+                action()
+            except BaseException as error:
+                errors.append({'action': name, 'error': repr(error)})
+        for package in (HOST, PACKAGE):
+            if package in self.owned:

+        else:

+        if PACKAGE in self.owned:

+        for area, key in SETTING_KEYS:

+            attempt('restore ' + key, lambda parts=parts: self.call(*parts))
+        attempt('restore hwui', lambda: self.call('shell', 'setprop', PROPERTY, self.initial['hwui'] or "''"))
+        for package in (TEST_PACKAGE, PACKAGE, HOST):
+            attempt('uninstall ' + package, lambda package=package: self.uninstall(package))
+        if self.initial['focus'] == ORIGINAL_MAIN:
+            attempt('restore original MainActivity', lambda: self.call('shell', 'am', 'start', '-W', '-n', ORIGINAL_MAIN))
+        final = {}
+        if self.initial['focus'] == ORIGINAL_MAIN:
+            def verify_focus():
+                deadline = time.monotonic() + 5
+                while True:
+                    windows = self.call('shell', 'dumpsys', 'window', 'windows').stdout
+                    final['window_dump'] = windows
+                    focus = re.search(r'mCurrentFocus=Window\{[^\n]*?\s([A-Za-z0-9_.]+/[A-Za-z0-9_.]+)\}', windows)
+                    if focus and focus.group(1) == ORIGINAL_MAIN:
+                        final['focus'] = ORIGINAL_MAIN
+                        return
+                    if time.monotonic() >= deadline:
+                        raise RuntimeError('Original MainActivity did not regain focus')
+                    time.sleep(0.1)
+            attempt('verify original focus', verify_focus)
+        for area, key in SETTING_KEYS:
+            attempt('verify ' + key, lambda area=area, key=key: final.update({key: self.call('shell', 'settings', 'get', area, key).stdout.strip()}))
+        attempt('verify default IME', lambda: final.update(default_ime=self.call('shell', 'settings', 'get', 'secure', 'default_input_method').stdout.strip()))
+        attempt('verify enabled IMEs', lambda: final.update(enabled_imes=self.call('shell', 'settings', 'get', 'secure', 'enabled_input_methods').stdout.strip()))
+        attempt('verify hwui', lambda: final.update(hwui=self.call('shell', 'getprop', PROPERTY).stdout.strip()))
+        for key in ('default_ime', 'hwui', *(key for _, key in SETTING_KEYS)):
+            if final.get(key) != self.initial[key]:

+        if enabled_components(final.get('enabled_imes', '')) != enabled_components(self.initial['enabled_imes']):
+            errors.append({'action': 'verify enabled IMEs', 'error': 'Enabled IME set changed; unowned IMEs were not modified'})

+        # while its adb response timed out. Do not adopt/remove an uncertain package
+        # by name; report the residue and require identity verification before cleanup.
+        attempt('verify diagnostic packages absent', lambda: final.update(
+            diagnostic_packages=sorted(self.packages() & PACKAGES)))
+        remaining_diagnostic = final.get('diagnostic_packages')

+            errors.append({'action': 'verify diagnostic packages absent',
+                           'error': 'Initially absent diagnostic packages remain; unconfirmed apps were not adopted or removed',
+                           'packages': remaining_diagnostic})
+        elif remaining_diagnostic is None:
+            errors.append({'action': 'verify diagnostic packages absent',
+                           'error': 'Could not confirm that all initially absent diagnostic packages are absent again'})
+        attempt('restore and verify wakefulness', lambda: self.restore_wakefulness(final))
+        save(self.args.output / 'restoration.json', {'initial': self.initial, 'final': final,

+                                                  'remaining_diagnostic_packages': remaining_diagnostic,
+                                                  'install_attempts': self.install_attempts,
+                                                  'audit_errors': self.audit_errors, 'errors': errors})

+def make_plan(labels, mode, blocks, seed):
+    if len(labels) not in (2, 3, 4) or len(set(labels)) != len(labels):
+        raise ValueError('Choose two, three or four distinct product labels')
+    if any(not re.fullmatch(r'[a-z][a-z0-9_-]*', label) for label in labels):
+        raise ValueError('Product labels must be simple lowercase names')
+    if mode not in ('pilot', 'formal') or not 1 <= blocks <= 24:
+        raise ValueError('Invalid study mode or block count')
+    if mode == 'pilot':
+        if blocks != 1:
+            raise ValueError('A pilot uses exactly one block')
+        # The caller already selects the forward/reverse pilot order.
+        return [list(labels)]
+    rng = random.Random(seed)
+    if len(labels) == 2:
+        a, b = labels
+        if blocks % 2:
+            raise ValueError('This confirmation needs equal numbers of ABBA and BAAB blocks')
+        result = [[a, b, b, a], [b, a, a, b]] * (blocks // 2)
+        rng.shuffle(result)
+        return result
+    if len(labels) == 4:
+        if blocks % 4:
+            raise ValueError('A four-product screen needs complete Williams squares')
+        a, b, c, d = labels
+        rows = [[a,b,d,c], [b,c,a,d], [c,d,b,a], [d,a,c,b]]
+        result = []
+        for _ in range(blocks // 4):
+            group = [list(row) for row in rows]
+            rng.shuffle(group)
+            result.extend(group)
+        return result
+    if blocks % 6:
+        raise ValueError('A formal three-product study needs a multiple of six blocks')
+    # Each complete group of six blocks balances every product's ordinal position.
+    permutations = list(itertools.permutations(labels))
+    result = []
+    while len(result) < blocks:
+        rng.shuffle(permutations)
+        result.extend([list(order) for order in permutations])
+    return result[:blocks]

+def positive_ns(record, key):
+    value = record.get(key)
+    if type(value) is not int or value <= 0:
+        raise ValueError(f'{key} must be a positive integer timestamp in nanoseconds')
+    return value

+def measurements(data, expected_commit, expected_protocol=PROTOCOL):

+        raise ValueError('Invalid phase experiment condition')

+        raise ValueError('Driver did not run outside the IME process')
+    rounds = ime['rounds']
+    if len(rounds) != 6 or len(editor['commits']) != 6 or editor['text'] != '你' * 6:
+        raise ValueError('A sample must contain exactly six successful selections')
+    rows = []

+            'n_dispatch_down_to_i_dispatch_up_ms': (i['dispatch_up_ns'] - n['dispatch_down_ns']) / 1e6,
+            'select_planned_up_to_ic_ms': (ic - selected['planned_up_ns']) / 1e6,
+            'select_injected_up_to_ic_ms': (ic - selected['inject_up_ns']) / 1e6,
+            'select_dispatch_up_to_ic_ms': (ic - selected['dispatch_up_ns']) / 1e6,
+        })
+    first = {key: value for key, value in rows[0].items() if key != 'round'}
+    warm = {key: statistics.median(row[key] for row in rows[1:]) for key in first}

+def describe(values):
+    return {'n': len(values), 'median': statistics.median(values),
+            'min': min(values), 'max': max(values), 'values': values}

+def summarize(samples, reference):
+    by_product = {}
+    for label in sorted({sample['label'] for sample in samples}):
+        product = [sample for sample in samples if sample['label'] == label]
+        by_product[label] = {
+            phase: {metric: describe([sample['measurements'][phase][metric] for sample in product])
+                    for metric in product[0]['measurements'][phase]}
+            for phase in ('first', 'warm_process_median')
+        }
+    paired = {}
+    for label in by_product:
+        if label == reference:
+            continue
+        differences = {}
+        for block in sorted({sample['block'] for sample in samples}):
+            left = [sample for sample in samples if sample['block'] == block and sample['label'] == reference]
+            right = [sample for sample in samples if sample['block'] == block and sample['label'] == label]
+            if not left or not right:
+                continue
+            for metric in left[0]['measurements']['first']:
+                # An ABBA block is one paired observation, not four independent processes.
+                difference = (statistics.mean(sample['measurements']['first'][metric] for sample in right) -
+                              statistics.mean(sample['measurements']['first'][metric] for sample in left))
+                differences.setdefault(metric, []).append(difference)
+        paired[label] = {metric: describe(values) for metric, values in differences.items()}
+    return {'products': by_product, 'paired_block_deltas_vs_reference': paired}

+"""Touch-shape validation; legacy evidence remains readable with its recorded policy."""
+
+MINIMUM_ACTUAL_HOLD_POLICY = 'minimum_20ms_from_actual_injection_down'
+MINIMUM_HOLD_NS = 20_000_000
+
+
+def validate_touch_hold(ime):
+    policy = ime.get('touch_hold_policy')
+    if policy is None:
+        return False  # Legacy sample; its immutable harness hash identifies its semantics.
+    if policy != MINIMUM_ACTUAL_HOLD_POLICY:
+        raise ValueError('Unknown touch hold policy')
+    rounds = ime.get('rounds')
+    if not isinstance(rounds, list) or not rounds:
+        raise ValueError('Declared touch hold policy requires observed touch records')
+    for index, row in enumerate(rounds, 1):
+        for key in ('n', 'i', 'select'):
+            event = row.get(key, {})
+            down, up = event.get('inject_down_ns'), event.get('inject_up_ns')
+            if type(down) is not int or type(up) is not int or down <= 0 or up - down < MINIMUM_HOLD_NS:
+                raise ValueError(f'Round {index} {key}: actual hold was shorter than the declared 20ms minimum')
+    return True
