+package io.github.sandyyur.inputbench;
+
+import java.net.ServerSocket;
+import java.net.Socket;
+import java.net.InetAddress;
+import android.util.Log;
+import android.os.Process;
+import android.os.SystemClock;
+import android.view.InputDevice;
+import android.view.InputEvent;
+import android.view.MotionEvent;
+import java.io.BufferedReader;
+import java.io.InputStreamReader;
+import java.io.OutputStreamWriter;
+import java.io.PrintWriter;
+import java.lang.reflect.Method;
+import org.json.JSONArray;
+import org.json.JSONObject;
+
+/** Shell-only experiment driver: never loaded into the IME or shipped in its APK. */
+public final class BenchTouchDriver {
+    private static long now() { return SystemClock.elapsedRealtimeNanos(); }
+
+    private static void waitUntil(long time) {
+        for (;;) {
+            long remaining = time - now();
+            if (remaining <= 0) return;
+            SystemClock.sleep((remaining + 999_999L) / 1_000_000L);
+        }
+    }
+
+    public static void main(String[] args) throws Exception {
+        try { run(args); }
+        catch (Exception error) { Log.e("Pr32TouchDriver", "External input driver failed", error); throw error; }
+    }
+
+    private static void run(String[] args) throws Exception {
+        if (args.length != 1 || !args[0].matches("pr32-touch-[0-9]+"))
+            throw new IllegalArgumentException("Expected a dedicated experiment socket name");
+        Class<?> inputClass = Class.forName("android.hardware.input.InputManagerGlobal");
+        Object input = inputClass.getMethod("getInstance").invoke(null);
+        Method inject = inputClass.getMethod("injectInputEvent", InputEvent.class, int.class);
+        int port = 20000 + Integer.parseInt(args[0].substring("pr32-touch-".length())) % 20000;
+        ServerSocket server = new ServerSocket(port, 1, InetAddress.getByName("127.0.0.1"));
+        server.setSoTimeout(60_000);
+        Socket socket = null;
+        try {
+            socket = server.accept();
+            try (BufferedReader reader = new BufferedReader(new InputStreamReader(socket.getInputStream(), "UTF-8"));
+             PrintWriter writer = new PrintWriter(new OutputStreamWriter(socket.getOutputStream(), "UTF-8"), true)) {
+            writer.println(new JSONObject().put("pid", Process.myPid()).put("uid", Process.myUid())
+                .put("protocol", "external_input_manager_v1").put("token", args[0]).put("transport", "loopback_tcp")
+                .put("driver_version", 2).put("actual_hold_min_ns", 20_000_000L));
+            String line;
+            while ((line = reader.readLine()) != null) {
+                JSONObject request = new JSONObject(line);
+                JSONArray touches = request.getJSONArray("touches");
+                if (touches.length() < 1 || touches.length() > 2) throw new IllegalArgumentException("Invalid batch");
+                JSONArray results = new JSONArray();
+                for (int index = 0; index < touches.length(); index++) {
+                    JSONObject touch = touches.getJSONObject(index);
+                    long plannedDown = touch.getLong("planned_down_ns");
+                    long plannedUp = touch.getLong("planned_up_ns");
+                    if (plannedUp - plannedDown != 20_000_000L) throw new IllegalArgumentException("Invalid hold");
+                    long downTime = touch.getLong("down_time_ms");
+                    JSONObject result = new JSONObject().put("down_time_ms", downTime).put("key", touch.getString("key"));
+                    long actualDown = 0;
+                    for (int action : new int[]{MotionEvent.ACTION_DOWN, MotionEvent.ACTION_UP}) {
+                        // A late DOWN must not compress the intended physical hold to zero.
+                        waitUntil(action == MotionEvent.ACTION_DOWN ? plannedDown : Math.max(plannedUp, actualDown + 20_000_000L));
+                        MotionEvent event = MotionEvent.obtain(downTime, SystemClock.uptimeMillis(), action,
+                            (float)touch.getDouble("x"), (float)touch.getDouble("y"), 0);
+                        event.setSource(InputDevice.SOURCE_TOUCHSCREEN);
+                        boolean down = action == MotionEvent.ACTION_DOWN;
+                        long timestamp = now();
+                        if (down) actualDown = timestamp;
+                        result.put(down ? "inject_down_ns" : "inject_up_ns", timestamp);
+                        try {
+                            if (!Boolean.TRUE.equals(inject.invoke(input, event, 0)))
+                                throw new IllegalStateException("Input injection rejected");
+                        } finally { event.recycle(); }
+                        result.put(down ? "inject_down_return_ns" : "inject_up_return_ns", now());
+                    }
+                    results.put(result);
+                }
+                writer.println(new JSONObject().put("touches", results));
+                if (writer.checkError()) throw new IllegalStateException("Experiment client disconnected");
+            }
+            }
+        } finally {
+            if (socket != null) socket.close();
+            server.close();
+        }
+    }
+}

+#!/usr/bin/env python3
+"""Validate the complete four-product screen and apply the predeclared preference."""
+import argparse
+import importlib.util
+import json

+
+HERE = Path(__file__).resolve().parent
+spec = importlib.util.spec_from_file_location('reviewed_study', HERE / 'run-study.py')

+def load_cohort(folder, sources, labels, seed, blocks):
+    files = list(folder.rglob('samples.json'))
+    if len(files) != 1:
+        raise ValueError('Expected exactly one complete original-configuration cohort')
+    path = files[0]
+    samples = json.loads(path.read_text())
+    expected_plan = study.make_plan(labels, 'formal', blocks, seed)
+    actual_plan = json.loads((path.parent / 'study-plan.json').read_text())
+    if (actual_plan['seed'] != seed or actual_plan['plan'] != expected_plan or actual_plan['profile'] or
+            actual_plan['sources'] != sources or len(samples) != sum(map(len, expected_plan))):
+        raise ValueError('The completed cohort differs from its frozen source/order/count plan')
+    expected_positions = {(i + 1, j + 1, label) for i, block in enumerate(expected_plan)
+                          for j, label in enumerate(block)}
+    if {(s['block'], s['position'], s['label']) for s in samples} != expected_positions:
+        raise ValueError('Duplicate or missing process position')
+    for sample in samples:
+        raw = json.loads((path.parent / sample['raw']).read_text())
+        if (sample['source'] != sources[sample['label']] or raw['ime']['letter_gap_ms'] != 100 or
+                raw['ime']['comment_mode'] != 'visible'):
+            raise ValueError('Unexpected source or stimulus condition')
+        if study.measurements(raw, sample['source']) != sample['measurements']:
+            raise ValueError('Raw measurement replay disagrees')
+    statistics_ = study.summarize(samples, 'baseline')
+    if json.loads((path.parent / 'summary.json').read_text())['statistics'] != statistics_:
+        raise ValueError('Archived summary differs from verified samples')
+    # Also aggregate first and within-process warm values independently by block.
+    paired = {}
+    for label in labels:
+        if label == 'baseline':
+            continue
+        paired[label] = {}
+        for phase in ('first', 'warm_process_median'):
+            rows = {}
+            for block in range(1, blocks + 1):
+                base = [s for s in samples if s['block'] == block and s['label'] == 'baseline']
+                target = [s for s in samples if s['block'] == block and s['label'] == label]
+                for metric in base[0]['measurements'][phase]:
+                    difference = statistics.mean(s['measurements'][phase][metric] for s in target) - statistics.mean(s['measurements'][phase][metric] for s in base)
+                    rows.setdefault(metric, []).append(difference)
+            paired[label][phase] = {metric: dict(n=len(values), median=statistics.median(values),
+                min=min(values), max=max(values), positive=sum(v > 0 for v in values),
+                negative=sum(v < 0 for v in values), values=values) for metric, values in rows.items()}
+    return dict(processes=len(samples), selections=len(samples) * 6, statistics=statistics_, paired=paired)
+
+
+def main():
+    parser = argparse.ArgumentParser(description=__doc__)
+    parser.add_argument('--forward', type=Path, required=True)
+    parser.add_argument('--reverse', type=Path, required=True)
+    parser.add_argument('--quality', type=Path, required=True)
+    parser.add_argument('--output', type=Path, required=True)
+    args = parser.parse_args()
+    plan = json.loads((HERE / 'reapply-plan.json').read_text())
+    quality = json.loads(args.quality.read_text())
+    if quality['status'] != 'pass' or quality['failures_errors_skipped'] != 0 or len(quality['results']) != 4:
+        raise ValueError('Every product must pass the complete device contract first')
+    cohorts = [load_cohort(folder, plan['sources'], tuple(order.split(',')), seed, 4)
+               for folder, order, seed in zip((args.forward, args.reverse), plan['screen']['orders'], plan['screen']['seeds'])]
+    excluded = {}
+    for label in plan['selection']['candidate_preference']:
+        excluded[label] = [metric for metric in plan['selection']['cold_metrics'] if all(
+            c['paired'][label]['first'][metric]['median'] > 0 and
+            c['paired'][label]['first'][metric]['positive'] >= 3 for c in cohorts)]
+    selected = next((label for label in plan['selection']['candidate_preference'] if not excluded[label]), 'none')
+    args.output.mkdir(parents=True, exist_ok=True)
+    result = dict(candidate=selected, sources=plan['sources'], exclusion_reasons=excluded,
+                  rule=plan['selection'], cohorts=cohorts,
+                  purpose='Choose the next independent test, not a product promotion or speedup claim')
+    (args.output / 'selection.json').write_text(json.dumps(result, indent=2) + '\n')
+    if os.environ.get('GITHUB_OUTPUT'):
+        with open(os.environ['GITHUB_OUTPUT'], 'a') as out:
+            out.write('candidate=' + selected + '\n')
+    print(json.dumps(dict(candidate=selected, exclusion_reasons=excluded)))
+
+
+if __name__ == '__main__':
+    main()

+"""Raw AndroidJUnitRunner status parser, shared by isolated quality checks."""
+def parse_status(text):
+    """Parse AndroidJUnitRunner raw status packets, including stack continuations."""
+    events, packet, last_key = [], {}, None
+    for line in text.splitlines():
+        if line.startswith('INSTRUMENTATION_STATUS: '):
+            field = line.removeprefix('INSTRUMENTATION_STATUS: ')
+            key, sep, value = field.partition('=')
+            if sep:
+                packet[key], last_key = value, key
+        elif line.startswith('INSTRUMENTATION_STATUS_CODE: '):
+            code = int(line.removeprefix('INSTRUMENTATION_STATUS_CODE: '))
+            if code != 1 and packet.get('test') and packet.get('class'):
+                events.append(dict(packet, status_code=code))
+            packet, last_key = {}, None
+        elif line.startswith('INSTRUMENTATION_'):
+            last_key = None
+        elif last_key:
+            packet[last_key] += '\n' + line
+    if 'INSTRUMENTATION_CODE: -1' not in text or 'INSTRUMENTATION_FAILED:' in text:
+        raise ValueError('The instrumentation process did not finish normally')
+    if not events:
+        raise ValueError('No executed test case results')
+    return events
