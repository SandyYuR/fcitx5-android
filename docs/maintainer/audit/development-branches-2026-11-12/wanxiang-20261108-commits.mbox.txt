+                    override fun dispatchTouchEvent(event: MotionEvent): Boolean {
+                        if (event.actionMasked == MotionEvent.ACTION_DOWN) dispatchDown = now()
+                        if (event.actionMasked == MotionEvent.ACTION_UP) dispatchUp = now()

+            }

+                    event.source = InputDevice.SOURCE_TOUCHSCREEN
+                    if (action == MotionEvent.ACTION_UP) injectUp = now()

+                    if (action == MotionEvent.ACTION_DOWN) SystemClock.sleep(20)
+                }
+                check(dispatchUp > 0 && dispatchReturned >= dispatchUp) { "Touch did not reach IME window" }
+                return JSONObject().put("key", text).put("inject_up_ns", injectUp)
+                    .put("dispatch_down_ns", dispatchDown).put("dispatch_up_ns", dispatchUp).put("dispatch_return_ns", dispatchReturned)
+            }
+            val rounds = JSONArray()
+            repeat(6) { round ->

+                    .put("preedit_collected_ns", preedit).put("commit_collected_ns", commit.second))
+            }
+            assertNull(withTimeoutOrNull(250) { commits.receive() })
+            val maps = File("/proc/self/maps").readText()
+            val labels = connection.runOnReady { statusArea().map { it.shortText } }

+            assertTrue(maps.contains("wanxiang-lts-zh-hans.gram"))
+            File(ctx.filesDir, "candidate-taps.json").writeText(JSONObject().put("commit", BuildConfig.BUILD_GIT_HASH)

+                .put("model_mapped", true).put("fcitx_core_mapped", maps.contains("libFcitx5Core.so"))

+            ins.runOnMainSync { wrappedWindow?.callback = oldCallback }
+            collector.cancelAndJoin()
+            FcitxDaemon.disconnect(javaClass.name)
+        }
+    }

+        for (int action = 0; action < 2; action++) {
+            MotionEvent event = MotionEvent.obtain(down, SystemClock.uptimeMillis(), action,
+                (float) point.getDouble(0), (float) point.getDouble(1), 0);
+            event.setSource(InputDevice.SOURCE_TOUCHSCREEN);
+            try {
+                times[action] = now();
+                if (!getUiAutomation().injectInputEvent(event, true)) throw new IllegalStateException("Touch injection failed");
+            } finally { event.recycle(); }
+            if (action == MotionEvent.ACTION_DOWN) SystemClock.sleep(24);

+public final class FileTimes {
+    public static void main(String[] args) throws Exception {
+        if(Process.myUid()!=0)throw new IllegalStateException("Disposable root emulator required");
+        if(args[0].equals("capture")) {
+            JSONArray rows=new JSONArray();
+            for(int i=1;i<args.length-1;i++) {
+                try(Stream<Path> paths=Files.walk(Paths.get(args[i]))) {
+                    for(Iterator<Path> it=paths.iterator();it.hasNext();) {
+                        Path p=it.next();if(Files.isSymbolicLink(p))continue;
+                        rows.put(new JSONObject().put("path",p.toString())
+                            .put("mtime_ns",Long.toString(Files.getLastModifiedTime(p).to(TimeUnit.NANOSECONDS))));
+                    }
+                }
+            }
+            Files.write(Paths.get(args[args.length-1]),rows.toString(2).getBytes("UTF-8"));
+            System.out.println("CAPTURED "+rows.length()+" exact file times");
+        } else if(args[0].equals("restore")) {
+            JSONArray rows=new JSONArray(new String(Files.readAllBytes(Paths.get(args[1])),"UTF-8"));
+            // Children before parents, so directory mtimes remain those captured.
+            for(int i=rows.length()-1;i>=0;i--) {
+                JSONObject row=rows.getJSONObject(i);Path p=Paths.get(row.getString("path"));
+                long wanted=Long.parseLong(row.getString("mtime_ns"));
+                if(!Files.exists(p)||Files.isSymbolicLink(p))throw new IllegalStateException("Fixture path missing: "+p);
+                Files.setLastModifiedTime(p,FileTime.from(wanted,TimeUnit.NANOSECONDS));
+                if(Files.getLastModifiedTime(p).to(TimeUnit.NANOSECONDS)!=wanted)
+                    throw new IllegalStateException("Nanosecond timestamp restoration failed: "+p);
+            }
+            System.out.println("RESTORED "+rows.length()+" exact file times");
+        } else throw new IllegalArgumentException("capture or restore");
+    }
