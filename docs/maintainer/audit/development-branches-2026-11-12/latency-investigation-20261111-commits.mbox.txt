+"""Summarize per-event trace associations; never merge with untraced benchmarks."""
+import argparse
+import csv
+import json
+from pathlib import Path
+import statistics
+
+
+def describe(values):
+    return dict(n=len(values),median=statistics.median(values),min=min(values),max=max(values))
+
+
+def load(path,expected):
+    data=json.loads(path.read_text())
+    assert not data['missing_native_anchors'],path
+    assert not data['trace_stats'],path
+    assert len(data['anchors'])==expected,path
+    assert all(row['core_calls']==1 for row in data['anchors']),path
+    return data
+
+
+def repeat(data):
+    result={}
+    for phase in ['fresh','rebind']:
+        rows=[r for r in data['anchors'] if r['phase']==phase]
+        result[phase]={metric:describe([r[metric] for r in rows]) for metric in
+            ['total_up_ms','core_wall_ms','core_cpu_ms','snapshot_ms']}
+        result[phase]['selected_rounds']={str(index):next(r for r in rows if r['round']==index)
+            for index in [1,4,8,16]}
+        result[phase]['max_core']=max(rows,key=lambda r:r['core_wall_ms'])
+        result[phase]['max_total']=max(rows,key=lambda r:r['total_up_ms'])
+    return result
+
+
+def main():
+    parser=argparse.ArgumentParser(description=__doc__)
+    parser.add_argument('--evidence',type=Path,required=True)
+    parser.add_argument('--output',type=Path,required=True)
+    args=parser.parse_args();root=args.evidence
+    candidates=[]
+    for index in range(1,4):
+        folder=root/f'candidate-analysis/process-{index}'
+        data=load(folder/'associated.json',12)
+        rows=[r for r in data['anchors'] if r['kind']=='n']
+        system=list(csv.DictReader((folder/'first-core-system-sched.csv').open()))
+        waits=[]
+        for wait in data['input_waits']:
+            if wait['wait_ms']<=8:continue
+            overlap=list(csv.DictReader((folder/f"input-wait-r{wait['round']}-{wait['key']}.csv").open()))
+            waits.append(dict(wait,overlap=overlap))
+        candidates.append(dict(process=index,first_n=rows[0],warm_n_cpu=describe([r['core_cpu_ms'] for r in rows[1:]]),
+            first_n_system_cpu=system,input_waits_over_8ms=waits,
+            select_inject_to_dispatch=describe([r['wait_ms'] for r in data['input_waits'] if r['key']=='select'])))
+    result={'method':'Traced attribution only; individual ordered timestamps, not subtracted percentiles.',
+        'candidate':candidates,'debug_repeat':{case:repeat(load(root/f'repeat-analysis/{case}/associated.json',32))
+            for case in ['ni','phrase']}}
+    release=root/'release-analysis/associated.json'
+    if release.exists():result['release_control']=repeat(load(release,32))
+    args.output.write_text(json.dumps(result,ensure_ascii=False,indent=2)+'\n')
+    print(json.dumps({'candidate_anchors':36,'debug_repeat_anchors':64,
+        'release_control_anchors':32 if release.exists() else 0,'trace_loss':0}))
+
+
+if __name__=='__main__':main()
